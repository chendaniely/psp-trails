Routes

From the Park Centre, Cleveland Trail at W 16th Ave

Pick a distance and a route type. Everyone who opens this page on the same day, with the same choices, gets the same route of the day; Another route shuffles. Routes stay on park trails as much as possible and use streets only to link trails up.

function rangeSlider({min, max, step, value, label}) {
  const form = html`<form class="range2">
    <label>${label}</label>
    <div class="range2-track">
      <input type="range" name="lo" min=${min} max=${max} step=${step} value=${value[0]} aria-label="Shortest">
      <input type="range" name="hi" min=${min} max=${max} step=${step} value=${value[1]} aria-label="Longest">
    </div>
    <output name="out"></output>
  </form>`;
  const {lo, hi, out} = form.elements;
  const track = form.querySelector(".range2-track");
  const pct = (v) => `${(100 * (v - min)) / (max - min)}%`;
  const update = () => {
    form.value = [+lo.value, +hi.value];
    out.value = `${lo.value}–${hi.value} km`;
    track.style.setProperty("--lo", pct(lo.value));
    track.style.setProperty("--hi", pct(hi.value));
  };
  form.addEventListener("input", (event) => {
    // Thumbs can't pass each other.
    if (+lo.value > +hi.value) {
      if (event.target === lo) lo.value = hi.value;
      else hi.value = lo.value;
    }
    update();
  });
  form.addEventListener("submit", (event) => event.preventDefault());
  update();
  return form;
}

// FNV-1a string hash: turns the date into a number, the same on every phone.
function hash(text) {
  let h = 2166136261;
  for (const c of text) h = Math.imul(h ^ c.charCodeAt(0), 16777619);
  return h >>> 0;
}

// mulberry32: random numbers from a seed. Same seed, same sequence, anywhere.
function seeded(seed) {
  return () => {
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// Today's order for the matching routes: a Fisher-Yates shuffle seeded by the
// date. Everyone gets the same order; "Another route" steps along it, so no
// route repeats until you've seen them all.
order = {
  const random = seeded(hash(today));
  const idx = matching.map((_, i) => i);
  for (let i = idx.length - 1; i > 0; i--) {
    const j = Math.floor(random() * (i + 1));
    [idx[i], idx[j]] = [idx[j], idx[i]];
  }
  return idx;
}

function gpx(f) {
  const points = f.geometry.coordinates
    .map(([lon, lat]) => `<trkpt lat="${lat}" lon="${lon}"></trkpt>`)
    .join("\n      ");
  return `<?xml version="1.0" encoding="UTF-8"?>
<gpx version="1.1" creator="psp-trails" xmlns="http://www.topografix.com/GPX/1/1">
  <trk>
    <name>PSP ${f.properties.id}: ${f.properties.name} (${f.properties.km} km)</name>
    <trkseg>
      ${points}
    </trkseg>
  </trk>
</gpx>`;
}

// Share: the phone's share sheet (messages, WhatsApp, email, Strava, ...).
// The text has a link that opens this exact route; the GPX goes along where
// the phone allows files. With no share sheet, the text goes to the clipboard.
function shareRoute(f) {
  const p = f.properties;
  const link = `${location.origin}${location.pathname}?route=${p.key}`;
  const text = [
    `${p.name}: ${p.km.toFixed(1)} km ${p.type}, ↑${p.gain_m} m ↓${p.loss_m} m,`,
    "from the Park Centre (Cleveland Trail & W 16th Ave)",
    link,
    "",
    ...p.directions.map(([at, full]) => `${at.toFixed(1)} km  ${full}`)
  ].join("\n");
  const title = `Pacific Spirit run: ${p.name}`;
  const file = new File([gpx(f)], `psp-${p.id}.gpx`, {type: "application/gpx+xml"});
  const quietly = (e) => { if (e.name !== "AbortError") throw e; }; // user cancelled
  if (navigator.canShare?.({files: [file]})) {
    return navigator.share({title, text, files: [file]}).catch(quietly);
  }
  if (navigator.share) return navigator.share({title, text}).catch(quietly);
  return navigator.clipboard.writeText(text).then(() => "copied");
}

function shareButton(f) {
  const button = html`<button class="btn btn-quarto" type="button">Share</button>`;
  button.onclick = async () => {
    if ((await shareRoute(f)) === "copied") {
      button.textContent = "Copied ✓";
      setTimeout(() => (button.textContent = "Share"), 2000);
    }
  };
  return button;
}

function esc(text) {
  return String(text).replace(/[&<>"]/g, (c) => ({"&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;"})[c]);
}

function outline(f, size) {
  // Route shape, north up, scaled to fit a square.
  const pts = f.geometry.coordinates;
  const k = Math.cos((pts[0][1] * Math.PI) / 180);
  const xs = pts.map(([lon]) => lon * k), ys = pts.map(([, lat]) => lat);
  const [x0, x1, y0, y1] = [Math.min(...xs), Math.max(...xs), Math.min(...ys), Math.max(...ys)];
  const scale = (size - 12) / Math.max(x1 - x0, y1 - y0);
  const dx = (size - (x1 - x0) * scale) / 2, dy = (size - (y1 - y0) * scale) / 2;
  const P = pts.map((_, i) => [dx + (xs[i] - x0) * scale, size - dy - (ys[i] - y0) * scale]);
  const line = P.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
  const i = Math.floor(P.length * 0.08); // an arrow early on shows the direction
  const [ax, ay] = P[i], [bx, by] = P[Math.min(i + 3, P.length - 1)];
  const angle = (Math.atan2(by - ay, bx - ax) * 180) / Math.PI;
  return `<svg class="pc-outline" viewBox="0 0 ${size} ${size}" width="${size}" height="${size}">
    <polyline points="${line}" fill="none" stroke="#0b0b0b" stroke-width="2" stroke-linejoin="round"/>
    <path d="M -5 -4 L 4 0 L -5 4 z" fill="#0b0b0b" transform="translate(${bx.toFixed(1)} ${by.toFixed(1)}) rotate(${angle.toFixed(0)})"/>
    <circle cx="${P[0][0].toFixed(1)}" cy="${P[0][1].toFixed(1)}" r="4" fill="#fff" stroke="#0b0b0b" stroke-width="2"/>
    <text x="${size - 2}" y="10" text-anchor="end" font-size="9" font-weight="700">N↑</text>
  </svg>`;
}

// Shrink the text until the directions fit on the card (no hidden overflow).
function fitCards(root, start) {
  const cards = [...root.querySelectorAll(".pc")];
  const card = cards[0], steps = card.querySelector(".pc-steps");
  const over = () =>
    card.scrollHeight > card.clientHeight + 1 ||
    steps.scrollHeight > steps.clientHeight + 1 ||
    steps.scrollWidth > steps.clientWidth + 1;
  let size = start;
  const apply = () => cards.forEach((c) => (c.style.fontSize = `${size}pt`));
  apply();
  while (over() && size > 5) {
    size -= 0.25;
    apply();
  }
  return size;
}

function printCards(f, layout) {
  const page = `<!doctype html><html><head><meta charset="utf-8">
    <title>PSP ${esc(f.properties.id)} cards</title>
    <style>${CARD_CSS}
      @page { size: letter; margin: 0.3in; }
      body { margin: 0; }
      .sheet { display: grid; width: 7.9in;
               grid-template-columns: repeat(${layout.cols}, ${layout.w}in);
               grid-template-rows: repeat(${layout.rows}, ${layout.h}in); }
    </style></head>
    <body><div class="sheet">${cardHtml(f, layout).repeat(layout.n)}</div>
    <script>${fitCards.toString()}
      window.onload = () => { fitCards(document, ${layout.font}); window.print(); };
    <\/script></body></html>`;
  const w = window.open("", "_blank");
  w.document.write(page);
  w.document.close();
}
