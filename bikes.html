Bikes

Rides from the Park Centre on the trails open to bikes

Rides on every trail in the park that’s open to bikes: all but the hiking-only ones. Like the route of the day, they start and finish at the Park Centre (Cleveland Trail at W 16th Ave) and use streets only to link trails up, at bike distances. Everyone who opens this page on the same day, with the same choices, gets the same ride of the day; Another route shuffles; Routes lists them all. Where a ride crosses a street on a crosswalk, walk your bike.

function rangeSlider({min, max, step, value, label}) {
  const form = html`<form class="range2">
    <label>${label}</label>
    <div class="range2-track">
      <input type="range" name="lo" min=${min} max=${max} step=${step} value=${value[0]} aria-label="Shortest">
      <input type="range" name="hi" min=${min} max=${max} step=${step} value=${value[1]} aria-label="Longest">
    </div>
    <output name="out"></output>
  </form>`;
  const {lo, hi, out} = form.elements;
  const track = form.querySelector(".range2-track");
  const pct = (v) => `${(100 * (v - min)) / (max - min)}%`;
  const update = () => {
    form.value = [+lo.value, +hi.value];
    out.value = `${lo.value}–${hi.value} km`;
    track.style.setProperty("--lo", pct(lo.value));
    track.style.setProperty("--hi", pct(hi.value));
  };
  form.addEventListener("input", (event) => {
    // Thumbs can't pass each other.
    if (+lo.value > +hi.value) {
      if (event.target === lo) lo.value = hi.value;
      else hi.value = lo.value;
    }
    update();
  });
  form.addEventListener("submit", (event) => event.preventDefault());
  update();
  return form;
}

function outline(f, size) {
  // Route shape, north up, scaled to fit a square.
  const pts = f.geometry.coordinates;
  const k = Math.cos((pts[0][1] * Math.PI) / 180);
  const xs = pts.map(([lon]) => lon * k), ys = pts.map(([, lat]) => lat);
  const [x0, x1, y0, y1] = [Math.min(...xs), Math.max(...xs), Math.min(...ys), Math.max(...ys)];
  const scale = (size - 12) / Math.max(x1 - x0, y1 - y0);
  const dx = (size - (x1 - x0) * scale) / 2, dy = (size - (y1 - y0) * scale) / 2;
  const P = pts.map((_, i) => [dx + (xs[i] - x0) * scale, size - dy - (ys[i] - y0) * scale]);
  const line = P.map(([x, y]) => `${x.toFixed(1)},${y.toFixed(1)}`).join(" ");
  const i = Math.floor(P.length * 0.08); // an arrow early on shows the direction
  const [ax, ay] = P[i], [bx, by] = P[Math.min(i + 3, P.length - 1)];
  const angle = (Math.atan2(by - ay, bx - ax) * 180) / Math.PI;
  return `<svg class="pc-outline" viewBox="0 0 ${size} ${size}" width="${size}" height="${size}">
    <polyline points="${line}" fill="none" stroke="#0b0b0b" stroke-width="2" stroke-linejoin="round"/>
    <path d="M -5 -4 L 4 0 L -5 4 z" fill="#0b0b0b" transform="translate(${bx.toFixed(1)} ${by.toFixed(1)}) rotate(${angle.toFixed(0)})"/>
    <circle cx="${P[0][0].toFixed(1)}" cy="${P[0][1].toFixed(1)}" r="4" fill="#fff" stroke="#0b0b0b" stroke-width="2"/>
    <text x="${size - 2}" y="10" text-anchor="end" font-size="9" font-weight="700">N↑</text>
  </svg>`;
}

// FNV-1a string hash: turns the date into a number, the same on every phone.
function hash(text) {
  let h = 2166136261;
  for (const c of text) h = Math.imul(h ^ c.charCodeAt(0), 16777619);
  return h >>> 0;
}

// mulberry32: random numbers from a seed. Same seed, same sequence, anywhere.
function seeded(seed) {
  return () => {
    seed = (seed + 0x6d2b79f5) | 0;
    let t = Math.imul(seed ^ (seed >>> 15), 1 | seed);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// Today's order for the matching routes: a Fisher-Yates shuffle seeded by the
// date. Everyone gets the same order; "Another route" steps along it, so no

  for (let i = idx.length - 1; i > 0; i--) {
    const j = Math.floor(random() * (i + 1));
    [idx[i], idx[j]] = [idx[j], idx[i]];
  }
  return idx;
}

function gpx(f) {
  const points = f.geometry.coordinates
    .map(([lon, lat]) => `<trkpt lat="${lat}" lon="${lon}"></trkpt>`)
    .join("\n      ");
  return `<?xml version="1.0" encoding="UTF-8"?>

  <trk>
    <name>PSP ${f.properties.id}: ${f.properties.name} (${f.properties.km} km)</name>
    <trkseg>
      ${points}
    </trkseg>
  </trk>
</gpx>`;
}

// Phones and tablets get the share sheet. Desktop browsers either have none
// or refuse (Chrome on a Mac: "Permission denied"), so there we copy or download.
touch = matchMedia("(pointer: coarse)").matches

async function shareSheet(data) {
  if (!touch || !navigator.canShare?.(data)) return "unavailable";
  try {
    await navigator.share(data);
    return "shared";
  } catch (e) {
    return e.name === "AbortError" ? "cancelled" : "failed";
  }
}

// Save an image: to the share sheet (Photos, Strava, ...) or as a download.
async function saveImage(blob, name, title) {
  const result = await shareSheet({files: [new File([blob], name, {type: "image/png"})], title});
  if (result === "shared" || result === "cancelled") return;
  const a = Object.assign(document.createElement("a"), {href: URL.createObjectURL(blob), download: name});
  a.click();
  setTimeout(() => URL.revokeObjectURL(a.href), 1000);
}

// Share: the phone's share sheet (messages, WhatsApp, email, Strava, ...).
// The text has a link that opens this exact route; the GPX goes along where
// the phone allows files. Otherwise the text goes to the clipboard.
async function shareRoute(f) {
  const p = f.properties;
  const link = `${location.origin}${location.pathname}?route=${p.key}`;
  const text = [
    `${p.name}: ${p.km.toFixed(1)} km ${p.type}, ↑${p.gain_m} m ↓${p.loss_m} m,`,
    "from the Park Centre (Cleveland Trail & W 16th Ave)",
    link,
    "",
    ...p.directions.map(([at, full]) => `${at.toFixed(1)} km  ${full}`)
  ].join("\n");
  const title = `Pacific Spirit run: ${p.name}`;
  const file = new File([gpx(f)], `psp-${p.id}.gpx`, {type: "application/gpx+xml"});
  let result = await shareSheet({title, text, files: [file]});
  if (result === "unavailable") result = await shareSheet({title, text});
  if (result === "shared" || result === "cancelled") return result;
  await navigator.clipboard.writeText(text);
  return "copied";
}

function shareButton(f) {
  const label = touch ? "Share" : "Copy link";
  const button = html`<button class="btn btn-quarto" type="button"
    title="The route's link, stats and directions, to paste into a message">${label}</button>`;
  button.onclick = async () => {
    const note = {copied: "Copied ✓", failed: "Couldn't copy"}[
      await shareRoute(f).catch(() => "failed")
    ];
    if (note) {
      button.textContent = note;
      setTimeout(() => (button.textContent = label), 2000);
    }
  };
  return button;
}

// Image: the route as a picture for Strava, Instagram or the group chat.
// 1080 x 1350 (4:5 portrait): the map as shown, name, stats, elevation, the
// trails in order, and credits.
async function routeImage(f) {
  const p = f.properties;
  const W = 1080, H = 1350, pad = 48;
  const font = (weight, size) => `${weight} ${size}px system-ui, -apple-system, "Segoe UI", sans-serif`;
  const canvas = Object.assign(document.createElement("canvas"), {width: W, height: H});
  const ctx = canvas.getContext("2d");
  ctx.fillStyle = "#fcfcfb";
  ctx.fillRect(0, 0, W, H);

  // The map, read straight after it draws (cover-fit into the top of the image).
  const snapshot = await new Promise((resolve) => {
    map.once("render", () => resolve(map.getCanvas().toDataURL("image/png")));
    map.triggerRepaint();
  });
  const img = new Image();
  await new Promise((resolve) => { img.onload = resolve; img.src = snapshot; });
  const mapH = 760, scale = Math.max(W / img.width, mapH / img.height);
  const [sw, sh] = [W / scale, mapH / scale];
  ctx.drawImage(img, (img.width - sw) / 2, (img.height - sh) / 2, sw, sh, 0, 0, W, mapH);

  // Name (shrunk to fit), stats and start.
  let y = mapH + 70, size = 52;
  ctx.fillStyle = "#0b0b0b";
  do { ctx.font = font(700, size); size -= 2; } while (ctx.measureText(p.name).width > W - 2 * pad);
  ctx.fillText(p.name, pad, y);
  ctx.font = font(600, 32);
  ctx.fillText(`${p.km.toFixed(1)} km ${p.type} · ↑${p.gain_m} m ↓${p.loss_m} m · ${p.trail_pct}% trail`, pad, y += 52);
  ctx.fillStyle = "#52514e";
  ctx.font = font(400, 24);
  ctx.fillText("Start/finish: Park Centre, Cleveland Trail & W 16th Ave · Pacific Spirit Regional Park", pad, y += 40);

  // Elevation line.
  const prof = p.profile, top = y + 24, h = 90;
  const [lo, hi] = [d3.min(prof, (d) => d[1]), d3.max(prof, (d) => d[1])];
  const px = (km) => pad + (km / p.km) * (W - 2 * pad);
  const py = (m) => top + h - ((m - lo) / Math.max(hi - lo, 1)) * h;
  ctx.beginPath();
  prof.forEach(([km, m], i) => (i ? ctx.lineTo(px(km), py(m)) : ctx.moveTo(px(km), py(m))));
  ctx.lineTo(px(p.km), top + h); ctx.lineTo(px(0), top + h); ctx.closePath();
  ctx.fillStyle = "rgba(42, 120, 214, 0.15)"; ctx.fill();
  ctx.beginPath();
  prof.forEach(([km, m], i) => (i ? ctx.lineTo(px(km), py(m)) : ctx.moveTo(px(km), py(m))));
  ctx.strokeStyle = "#2a78d6"; ctx.lineWidth = 3; ctx.stroke();

  // The trails, in running order.
  const unnamed = new Set(["unnamed trail", "crossing", "path", "sidewalk", "street"]);
  const trails = p.cues.map(([label]) => label.replace(/ Trail$/, ""))
    .filter((t, i, all) => !unnamed.has(t) && !t.startsWith("Cross ") && t !== all[i - 1]);
  ctx.fillStyle = "#0b0b0b";
  ctx.font = font(500, 26);
  let line = "", ty = top + h + 56;
  for (const word of trails.join(" › ").split(" ")) {
    if (ctx.measureText(`${line} ${word}`).width > W - 2 * pad && line) {
      ctx.fillText(line, pad, ty); ty += 36; line = word;
    } else line = line ? `${line} ${word}` : word;
    if (ty > H - 90) break;
  }
  if (ty <= H - 90) ctx.fillText(line, pad, ty);

  // Footer.
  ctx.fillStyle = "#52514e";
  ctx.font = font(400, 20);
  ctx.fillText(`pstrunners.github.io · route ${p.id}`, pad, H - 32);
  ctx.textAlign = "right";
  ctx.fillText("Map © OpenStreetMap contributors · OpenFreeMap", W - pad, H - 32);
  return new Promise((resolve) => canvas.toBlob(resolve, "image/png"));
}

function kmRange([a, b]) {
  const [from, to] = [Math.round(a), Math.round(b)];
  return from === to ? `km ${from}` : `km ${from}–${to}`;
}

// "start, km 19–21, km 51, finish" for the Park Centre; just the km elsewhere.
function aidPasses(name, stops) {
  const kms = stops.map(kmRange);
  return name === "Park Centre" ? ["start", ...kms.slice(1, -1), "finish"].join(", ") : kms.join(", ");
}

// Aid station passes (km), the sure ones or the optional ones.
function aidKm(optional) {
  return Object.entries(stats.aid)
    .filter(([name]) => stats.aid_optional.includes(name) === optional)
    .flatMap(([, stops]) => stops.map(([a]) => a));
}

// Labels for the passes mid-run, at the top of the profile.
function aidLabels(optional) {
  return aidKm(optional).filter((km) => km > 0 && km < stats.km).map((km) => ({km}));
}

function onTheWay(kind) {
  return stats.on_the_way.filter(([, , kinds]) => kinds.includes(kind)).map(kmRange);
}

// What's at this km, for the tooltip.
function whatsHere(km) {
  const near = (a, b = a) => a - 0.15 <= km && km <= b + 0.15;
  const what = Object.entries(stats.aid)
    .filter(([, stops]) => stops.some(([a, b]) => near(a, b)))
    .map(([name]) => `${stats.aid_optional.includes(name) ? "optional aid station" : "aid station"} (${name})`);
  what.push(...amenity.passedNear(stats.passes_by, km));
  return what.length ? ` · ${what.join(", ")}` : "";
}

function ultraGpx(name, coordinates) {
  const points = coordinates.map(([lon, lat]) => `<trkpt lat="${lat}" lon="${lon}"></trkpt>`);
  return `<?xml version="1.0" encoding="UTF-8"?>
<gpx version="1.1" creator="psp-trails" xmlns="http://www.topografix.com/GPX/1/1">
  <trk>
    <name>${name}</name>
    <trkseg>
      ${points.join("\n      ")}
    </trkseg>
  </trk>
</gpx>`;
}
